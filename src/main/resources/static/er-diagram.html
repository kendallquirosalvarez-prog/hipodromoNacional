        id_propietario, nombre, apellidos, id_barrio

CREATE OR REPLACE PROCEDURE sp_actualizar_caballo(
    p_id           VARCHAR,
    p_peso         DECIMAL(10,2),
    p_estado_salud VARCHAR,
    p_id_establo   VARCHAR
)
LANGUAGE plpgsql AS $$
BEGIN
    UPDATE caballo
    SET
        peso         = p_peso,
        estado_salud = p_estado_salud,
        id_establo   = p_id_establo
    WHERE id_caballo = p_id;
    -- El trigger actualiza bitacora_auditoria automáticamente
END;
$$;
sp_insertar_inscripcion INSERT 
CREATE OR REPLACE PROCEDURE sp_insertar_inscripcion(
    p_id          VARCHAR,
    p_id_evento   VARCHAR,
    p_id_caballo  VARCHAR,
    p_fecha       DATE,
    p_estado      VARCHAR
)
LANGUAGE plpgsql AS $$
BEGIN
    -- Validar que el caballo tenga certificación vigente
    IF NOT EXISTS (
        SELECT 1 FROM historial_veterinario
        WHERE id_caballo = p_id_caballo
          AND fecha_vencimiento_certificacion >= CURRENT_DATE
    ) THEN
        RAISE EXCEPTION 'Caballo sin certificación veterinaria vigente';
    END IF;

    INSERT INTO inscripcion (
        id_inscripcion, id_evento, id_caballo,
        fecha_inscripcion, estado
    )
    VALUES (
        p_id, p_id_evento, p_id_caballo,
        p_fecha, p_estado
    );
END;
$$;
sp_insertar_factura INSERT 
CREATE OR REPLACE PROCEDURE sp_insertar_factura(
    p_id             VARCHAR,
    p_id_propietario VARCHAR,
    p_id_evento      VARCHAR,
    p_subtotal       DECIMAL(12,2),
    p_descuento      DECIMAL(12,2),
    p_impuestos      DECIMAL(12,2),
    p_total          DECIMAL(12,2),
    p_estado_pago    VARCHAR
)
LANGUAGE plpgsql AS $$
BEGIN
    -- Verificar IVA 13% (legislación costarricense)
    IF ABS(p_impuestos - (p_subtotal - p_descuento) * 0.13) > 0.01 THEN
        RAISE EXCEPTION 'IVA debe ser 13%% del subtotal neto';
    END IF;

    INSERT INTO factura (
        id_factura, id_propietario, id_evento,
        subtotal, descuento, impuestos, total, estado_pago
    )
    VALUES (
        p_id, p_id_propietario, p_id_evento,
        p_subtotal, p_descuento, p_impuestos,
        p_total, p_estado_pago
    );
END;
$$;
tr_auditoria (Trigger) TRIGGER 
-- Función que alimenta la bitácora de auditoría
CREATE OR REPLACE FUNCTION fn_registrar_auditoria()
RETURNS TRIGGER LANGUAGE plpgsql AS $$
BEGIN
    INSERT INTO bitacora_auditoria (
        tabla_afectada, operacion,
        datos_anteriores, datos_nuevos, fecha_hora
    )
    VALUES (
        TG_TABLE_NAME,
        TG_OP,
        CASE WHEN TG_OP != 'INSERT'
             THEN row_to_json(OLD)::TEXT
             ELSE NULL END,
        CASE WHEN TG_OP != 'DELETE'
             THEN row_to_json(NEW)::TEXT
             ELSE NULL END,
        NOW()
    );
    RETURN NEW;
END;
$$;

-- Asociar el trigger a cada tabla principal
CREATE TRIGGER tr_auditoria_caballo
AFTER INSERT OR UPDATE OR DELETE ON caballo
FOR EACH ROW EXECUTE FUNCTION fn_registrar_auditoria();
